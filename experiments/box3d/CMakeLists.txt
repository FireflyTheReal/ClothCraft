cmake_minimum_required(VERSION 3.22)
project(clothcraft_box3d_lab LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BOX3D_SAMPLES OFF CACHE BOOL "" FORCE)
set(BOX3D_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(BOX3D_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BOX3D_DOCS OFF CACHE BOOL "" FORCE)
if(NOT DEFINED BOX3D_SOURCE)
  message(FATAL_ERROR "Pass -DBOX3D_SOURCE=<pinned checkout>; see docs/solver-lab.md")
endif()
add_subdirectory("${BOX3D_SOURCE}" box3d)
set(CLOTH_LAB_MESH_TRIANGLES 4096 CACHE STRING "Native mesh contact query capacity; stock Box3D is 512")
target_compile_definitions(box3d PRIVATE B3_MAX_MESH_CONTACT_TRIANGLES=${CLOTH_LAB_MESH_TRIANGLES})
add_executable(clothcraft_box3d main.cpp)
target_compile_features(clothcraft_box3d PRIVATE cxx_std_20)
target_link_libraries(clothcraft_box3d PRIVATE box3d::box3d)
